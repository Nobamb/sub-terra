            if (inventoryPanelRoot != null && inventoryPanelView == null)

                inventoryPanelRoot.SetActive(true);

                    inventoryPanelView.SetVisible(visible);
                }

            {
                inventoryPanelRoot.transform.SetAsLastSibling();
            }
        }
